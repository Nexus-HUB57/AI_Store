   221→   270→   270→   270→class AIPackageBuilder:
   222→   271→   271→   271→    def __init__(self, manifest_path: str):
   223→   272→   272→   272→        self.manifest_path = Path(manifest_path)
   224→   273→   273→   273→        self.package_dir = self.manifest_path.parent
   225→   274→   274→   274→        self.manifest_data = self._load_and_validate_manifest()
   226→   275→   275→   275→
   227→   276→   276→   276→    def _load_and_validate_manifest(self) -> Dict[str, Any]:
   228→   277→   277→   277→        if not self.manifest_path.exists():
   229→   278→   278→   278→            raise FileNotFoundError(f"Manifesto não encontrado em: {self.manifest_path}")
   230→   279→   279→   279→
   231→   280→   280→   280→        with open(self.manifest_path, "r", encoding="utf-8") as f:
   232→   281→   281→   281→            data = json.load(f)
   233→   282→   282→   282→
   234→   283→   283→   283→        required_keys = ["name", "version", "runtime", "a2a_interface"]
   235→   284→   284→   284→        for key in required_keys:
   236→   285→   285→   285→            if key not in data:
   237→   286→   286→   286→                raise ValueError(f"[Erro de Validação] Campo obrigatório ausente no manifesto: '{key}'")
   238→   287→   287→   287→
   239→   288→   288→   288→        return data
   240→   289→   289→   289→
   241→   290→   290→   290→    def build_package(self, output_file: str) -> str:
   242→   291→   291→   291→        """Empacota os arquivos do agente em um arquivo comprimido .aipkg (tar.gz)"""
   243→   292→   292→   292→        print(f"📦 Empacotando agente: {self.manifest_data['name']} (v{self.manifest_data['version']})...")
   244→   293→   293→   293→        
   245→   294→   294→   294→        with tarfile.open(output_file, "w:gz") as tar:
   246→   295→   295→   295→            # Adiciona o manifesto
   247→   296→   296→   296→            tar.add(self.manifest_path, arcname="aipkg.json")
   248→   297→   297→   297→            
   249→   298→   298→   298→            # Adiciona os arquivos do projeto evitando diretórios temporários
   250→   299→   299→   299→            for root, dirs, files in os.walk(self.package_dir):
   251→   300→   300→   300→                dirs[:] = [d for d in dirs if d not in [".git", "__pycache__", "node_modules", "dist"]]
   252→   301→   301→   301→                for file in files:
   253→   302→   302→   302→                    file_path = Path(root) / file
   254→   303→   303→   303→                    if file_path != Path(output_file) and file_path != self.manifest_path:
   255→   304→   304→   304→                        arcname = file_path.relative_to(self.package_dir)
   256→   305→   305→   305→                        tar.add(file_path, arcname=arcname)
   257→   306→   306→   306→
   258→   307→   307→   307→        # Gera o SHA-256 do pacote para integridade
   259→   308→   308→   308→        sha256_hash = hashlib.sha256()
   260→   309→   309→   309→        with open(output_file, "rb") as f:
   261→   310→   310→   310→            for byte_block in iter(lambda: f.read(4096), b""):
   262→   311→   311→   311→                sha256_hash.update(byte_block)
   263→   312→   312→   312→        
   264→   313→   313→   313→        digest = sha256_hash.hexdigest()
   265→   314→   314→   314→        print(f"✅ Pacote gerado com sucesso: {output_file}")
   266→   315→   315→   315→        print(f"🔒 SHA-256 Digest: {digest}")
   267→   316→   316→   316→        return digest

   272→   321→   321→   321→        self.api_token = api_token

   274→   323→   323→   323→    def publish_package(self, package_path: str, manifest: Dict[str, Any]) -> None:
   275→   324→   324→   324→        """Faz o upload do binário empacotado para o registro da AI Store"""
   276→   325→   325→   325→        agent_name = manifest["name"]
   277→   326→   326→   326→        version = manifest["version"]
   278→   327→   327→   327→        
   279→   328→   328→   328→        upload_endpoint = f"{self.registry_url}/packages/upload"
   280→   329→   329→   329→        print(f"🚀 Publicando {agent_name}@{version} para {upload_endpoint}...")
   281→   330→   330→   330→
   282→   331→   331→   331→        headers = {
   283→   332→   332→   332→            "Authorization": f"Bearer {self.api_token}",
   284→   333→   333→   333→            "Content-Type": "application/octet-stream",
   285→   334→   334→   334→            "X-Agent-Name": agent_name,
   286→   335→   335→   335→            "X-Agent-Version": version
   287→   336→   336→   336→        }
   288→   337→   337→   337→
   289→   338→   338→   338→        with open(package_path, "rb") as f:
   290→   339→   339→   339→            package_bytes = f.read()
   291→   340→   340→   340→
   292→   341→   341→   341→        req = urllib.request.Request(upload_endpoint, data=package_bytes, headers=headers, method="POST")
   293→   342→   342→   342→
   294→   343→   343→   343→        try:
   295→   344→   344→   344→            with urllib.request.urlopen(req) as response:
   296→   345→   345→   345→                if response.status in [200, 201]:
   297→   346→   346→   346→                    res_body = json.loads(response.read().decode("utf-8"))
   298→   347→   347→   347→                    print(f"🎉 Publicação concluída com sucesso!")
   299→   348→   348→   348→                    print(f"📍 URL de Acesso na AI Store: {res_body.get('package_url', 'N/A')}")
   300→   349→   349→   349→                else:
   301→   350→   350→   350→                    print(f"❌ Falha na publicação. Status Code: {response.status}")
   302→   351→   351→   351→        except urllib.error.HTTPError as e:
   303→   352→   352→   352→            print(f"❌ Erro de HTTP ao publicar: {e.code} - {e.read().decode('utf-8')}")
   304→   353→   353→   353→        except Exception as e:
   305→   354→   354→   354→            print(f"❌ Erro inesperado durante o upload: {str(e)}")

   760→   809→   809→   809→        # Otimização autônoma eleva o fitness score do aplicativo na Store
   761→   810→   810→   810→        mutated_app["fitnessScore"] = min(100.0, target_app.get("fitnessScore", 70.0) + 8.5)
